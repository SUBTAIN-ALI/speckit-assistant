# Ticket 15: File-Access Exclusions

**Priority:** P1  
**Depends On:** Ticket 04  
**Requirements Ref:** Section 7.2  
**Estimated Effort:** Small  

## Description

The chatbot agent must not access sensitive or irrelevant files during project investigation. While the AI CLI already has some sense of what to read, we must enforce exclusions for:

- **Secrets**: `.env`, `.env.local`, `.env.production`, `credentials.json`, `*.pem`, `*.key`, `*.p12`
- **Credential files**: `~/.ssh/`, `~/.aws/`, tokens, API keys in config files
- **Dependency folders**: `node_modules/`, `vendor/`, `.venv/`, `__pycache__/`
- **Build outputs**: `dist/`, `build/`, `.next/`, `out/`, `coverage/`
- **Large binary files**: images, videos, compiled binaries
- **Git internals**: `.git/` (but git CLI commands for history are allowed)
- **Spec Rabbit internals**: `.specify/.runtime/` (runtime state, not user content)

This exclusion list is injected into the agent's system prompt and, where possible, enforced at the file-access layer.

## Files to Create

### `src/chatbot/adapters/FileAccessGuard.ts`

```typescript
export const EXCLUDED_PATTERNS: string[] = [
  // Secrets and credentials
  '.env', '.env.*', '*.pem', '*.key', '*.p12', '*.pfx',
  'credentials.json', 'serviceAccountKey.json',
  '**/secrets/**',
  
  // Dependencies
  'node_modules/**', 'vendor/**', '.venv/**', '__pycache__/**',
  'bower_components/**', '.pnp.*',
  
  // Build outputs
  'dist/**', 'build/**', '.next/**', 'out/**', 'coverage/**',
  '.turbo/**', '.cache/**',
  
  // Git internals
  '.git/**',
  
  // Spec Rabbit runtime (not user content)
  '.specify/.runtime/**',
  
  // Large binaries
  '*.png', '*.jpg', '*.jpeg', '*.gif', '*.ico', '*.svg',
  '*.mp4', '*.mov', '*.avi',
  '*.woff', '*.woff2', '*.ttf', '*.eot',
  '*.zip', '*.tar', '*.gz', '*.jar', '*.war',
  '*.exe', '*.dll', '*.so', '*.dylib',
  
  // Lock files (large, not human-readable)
  'package-lock.json', 'yarn.lock', 'pnpm-lock.yaml',
  'Gemfile.lock', 'poetry.lock', 'Pipfile.lock',
];

export function buildExclusionPrompt(): string {
  return `
## File Access Restrictions
You MUST NOT read or access the following:
- Secret/credential files: .env, .env.*, *.pem, *.key, credentials.json
- Dependency directories: node_modules/, vendor/, .venv/
- Build output directories: dist/, build/, .next/, out/, coverage/
- Git internals: .git/ (use git CLI commands for history instead)
- Binary files: images, videos, fonts, archives, compiled binaries
- Lock files: package-lock.json, yarn.lock, pnpm-lock.yaml
- Spec Rabbit runtime: .specify/.runtime/

You CAN read:
- Source code files (*.ts, *.tsx, *.js, *.py, *.go, etc.)
- Documentation (*.md, *.txt, *.rst)
- Configuration (tsconfig.json, package.json [not lock], .eslintrc, etc.)
- Spec Rabbit data: .specify/memory/, .specify/meetings/, .specify/chatbot/knowledge/, specs/
`;
}
```

### Integration

- `ChatbotAgentRunner.spawn()` appends the exclusion prompt to the system prompt
- The exclusion prompt is generated by `buildExclusionPrompt()`
- If the host supports file-access restrictions at the agent runner level (e.g., passing an ignore list), use that too

## Acceptance Criteria

- [ ] Exclusion patterns defined covering all categories from Section 7.2
- [ ] Exclusion rules injected into agent system prompt
- [ ] `isExcluded()` function available for programmatic checks
- [ ] `.env` and credential files never surfaced in chatbot answers
- [ ] `node_modules/` and build outputs ignored
- [ ] Source code, docs, and Spec Rabbit data remain accessible

## Test Cases

### `src/chatbot/adapters/__tests__/FileAccessGuard.test.ts`

```typescript
import { isExcluded, EXCLUDED_PATTERNS, buildExclusionPrompt } from '../FileAccessGuard';

describe('FileAccessGuard', () => {
  // T15-1: .env excluded
  it('excludes .env files', () => {
    expect(isExcluded('.env')).toBe(true);
    expect(isExcluded('.env.local')).toBe(true);
    expect(isExcluded('.env.production')).toBe(true);
  });

  // T15-2: Credential files excluded
  it('excludes credential files', () => {
    expect(isExcluded('server.pem')).toBe(true);
    expect(isExcluded('private.key')).toBe(true);
    expect(isExcluded('credentials.json')).toBe(true);
  });

  // T15-3: node_modules excluded
  it('excludes node_modules', () => {
    expect(isExcluded('node_modules/express/index.js')).toBe(true);
  });

  // T15-4: Build outputs excluded
  it('excludes build output directories', () => {
    expect(isExcluded('dist/index.js')).toBe(true);
    expect(isExcluded('.next/server/app.js')).toBe(true);
    expect(isExcluded('build/static/main.js')).toBe(true);
  });

  // T15-5: Source code allowed
  it('allows source code files', () => {
    expect(isExcluded('src/auth/login.ts')).toBe(false);
    expect(isExcluded('lib/utils.py')).toBe(false);
  });

  // T15-6: Documentation allowed
  it('allows documentation files', () => {
    expect(isExcluded('README.md')).toBe(false);
    expect(isExcluded('docs/api.md')).toBe(false);
  });

  // T15-7: Spec Rabbit data allowed
  it('allows Spec Rabbit user data', () => {
    expect(isExcluded('.specify/memory/constitution.md')).toBe(false);
    expect(isExcluded('.specify/chatbot/knowledge/refund-policy.md')).toBe(false);
    expect(isExcluded('specs/auth/spec.md')).toBe(false);
  });

  // T15-8: Spec Rabbit runtime excluded
  it('excludes Spec Rabbit runtime', () => {
    expect(isExcluded('.specify/.runtime/workflow-state.json')).toBe(true);
  });

  // T15-9: Binary files excluded
  it('excludes binary files', () => {
    expect(isExcluded('logo.png')).toBe(true);
    expect(isExcluded('video.mp4')).toBe(true);
    expect(isExcluded('font.woff2')).toBe(true);
  });

  // T15-10: Lock files excluded
  it('excludes lock files', () => {
    expect(isExcluded('package-lock.json')).toBe(true);
    expect(isExcluded('yarn.lock')).toBe(true);
  });

  // T15-11: package.json allowed (not lock)
  it('allows package.json', () => {
    expect(isExcluded('package.json')).toBe(false);
  });

  // T15-12: Exclusion prompt is non-empty
  it('builds a non-empty exclusion prompt', () => {
    const prompt = buildExclusionPrompt();
    expect(prompt).toContain('.env');
    expect(prompt).toContain('node_modules');
    expect(prompt).toContain('CAN read');
  });

  // T15-13: .git directory excluded but git commands mentioned
  it('excludes .git directory', () => {
    expect(isExcluded('.git/HEAD')).toBe(true);
    expect(isExcluded('.git/objects/abc')).toBe(true);
  });
});
```
